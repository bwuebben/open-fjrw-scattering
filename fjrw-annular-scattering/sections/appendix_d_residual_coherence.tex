\section{Residual transports and products along different paths}
\label{app:residual-coherence}

Section~\ref{sec:gkt-decoration} compares $\mathfrak D$ with GKT
wall-crossing one ray at a time.  Multiplicative structures also involve
functions that reach one region along different paths: in the theta-function
constructions of
\cite{grosshackingkeel2015,grossetal2018canonical,grossetal2016theta},
structure constants come from pairs of broken lines that end at a common
point after travelling along different paths.  We do not define broken lines
on $\Acal^\circ$ (Section~\ref{subsec:core-holonomy}), but the same question
arises for admissible paths.  Along one path, transport through $\mathfrak D$
and transport by GKT's factors differ by one automorphism of the target chart
algebra, the \emph{residual transport} of the path
(Definition~\ref{def:gkt-transport}); being an algebra automorphism, it
relates the products of functions carried along that path.  Along different
paths the residual transports differ, and the product of the transports
through $\mathfrak D$ need not be the image of the product of the GKT
transports under any automorphism (see the example after
Definition~\ref{def:gkt-transport}).  As
a first step towards multiplying functions carried along different paths, we
correct each factor by a degree-weighted interpolation of the residual
transports, so that the corrected product is such an image, and we compute
the associator between parenthesizations.

\begin{definition}[Transports along a path]
\label{def:gkt-transport}
Fix a labelled truncation (Section~\ref{subsec:raywise-factorization}), a
deformation order, and a region of $\mathfrak D$
(Definition~\ref{def:Jacobi-wall}), the \emph{target region}, and work in the
corresponding finite nilpotent quotient.  Let $\gamma$ be a path transverse
to the walls that ends in the target region and realizes an admissible path
$\mathfrak p$ (Definition~\ref{def:framed-path}).
\begin{enumerate}
\item The \emph{transport through $\mathfrak D$} along $\gamma$ is
$P_\gamma=P_{\mathfrak p}$ (Section~\ref{subsec:KS-classes}), the composite,
in the order met, of the slab transitions and of the contributions
$\Theta_{\mathfrak d}^{\pm1}$ of the crossed walls
(Definition~\ref{def:Jacobi-wall}), from the chart algebra at the start of
$\gamma$ to that of the target region.
\item The \emph{GKT transport} $P^{\GKT}_\gamma$ is obtained from $P_\gamma$
by replacing, for every crossed wall $\mathfrak d$ whose factor lies in the
ray group of a shifted ray $\mathfrak r$, the contribution
$\Theta_{\mathfrak d}^{\pm1}$ by $\exp(\pm G_{\mathfrak r})$, with
$G_{\mathfrak r}$ the GKT logarithm of
Definition~\ref{def:raywise-logarithms}; at a chord, $\mathfrak r$ is the
chord ray, on either half of the chord.  The slab transitions are not
changed.
\item The \emph{residual transport} of $\gamma$ is the automorphism
$\mathcal Q_\gamma=P_\gamma(P^{\GKT}_\gamma)^{-1}$ of the chart algebra of
the target region, congruent to the identity modulo the ideal generated by
the $t_i$.
\end{enumerate}
\end{definition}

For the two labels of twists $(1,0)$ and $(0,1)$ of
Proposition~\ref{prop:first-decoration}, the paths $\gamma_\pm$ of
Proposition~\ref{prop:two-label-transport} lie in one chart and have the same
endpoints.  Both have transport $\Sigma_1\Sigma_2$ through $\mathfrak D$, but
by \eqref{eq:mixed-GKT-transition} and
Proposition~\ref{prop:two-label-transport}(3) their GKT transports are
$\Sigma_2Z_{1/4}\Sigma_1=Z_1\Sigma_1\Sigma_2$ and $\Sigma_1\Sigma_2$, with
$Z_c$ as in \eqref{eq:mixed-total-factor}.  Thus the GKT transport depends
on the path and not only on its endpoints, and the residual transports are
$Z_{-1}$ and the identity.  For $f_1,f_2$ in the chart algebra at the initial
point, the product of the transports through $\mathfrak D$ is $Z_{-1}(a)\,b$,
where $a=Z_1\Sigma_1\Sigma_2f_1$ and $b=\Sigma_1\Sigma_2f_2$ are the GKT
transports and run through the whole algebra.  Since $Z_{-1}\neq\mathrm{id}$,
this product is not the image of $ab$ under any automorphism independent of
$f_1,f_2$, by Proposition~\ref{prop:corridor-groupoid}.

\subsection{Corrected products}

Let $\gamma_1,\dots,\gamma_n$ be paths as in
Definition~\ref{def:gkt-transport}, let $f_i$ be a function in
the chart algebra at the initial point of $\gamma_i$, and write
$\mathcal Q_i=\mathcal Q_{\gamma_i}$, so that
$P_{\gamma_i}f_i=\mathcal Q_iP^{\GKT}_{\gamma_i}f_i$.  Rational powers of
automorphisms congruent to the identity are the finite polynomials
$g^c=\exp(c\log g)$, $c\in\QQ$, and for such $g,h$ we put
$g\mathbin{\#_c}h=g(g^{-1}h)^c$, which runs from $g$ at $c=0$ to $h$ at
$c=1$.  If $g$ and $h$ commute, then
$\log(g\mathbin{\#_c}h)=(1-c)\log g+c\log h$.  For $n=2$ the \emph{midpoint} of
the residual transports is $H=\mathcal Q_1\mathbin{\#_{1/2}}\mathcal Q_2$,
and the \emph{corrections} are $C_i=H\mathcal Q_i^{-1}$.

\begin{lemma}[Midpoint correction]\label{prop:residual-midpoint}
Let $g,h$ be as above and $c\in\QQ$.
\begin{enumerate}
\item $g\mathbin{\#_c}h=h\mathbin{\#_{1-c}}g$; in particular $H$ is symmetric
in $\mathcal Q_1,\mathcal Q_2$.
\item For $n=2$,
\begin{equation}\label{eq:corrected-product}
 C_iP_{\gamma_i}=HP^{\GKT}_{\gamma_i},
 \qquad
 (C_1P_{\gamma_1}f_1)(C_2P_{\gamma_2}f_2)
 =H\bigl((P^{\GKT}_{\gamma_1}f_1)(P^{\GKT}_{\gamma_2}f_2)\bigr).
\end{equation}
\item These constructions are natural under finite restriction.  If all paths
are continued by a common path from the target region, with transport $\Psi$
through $\mathfrak D$ and GKT transport $\Psi'$, then $\mathcal Q_i$ is
replaced by $\Psi\mathcal Q_i\Psi'^{-1}$, and
$(\Psi g\Psi'^{-1})\mathbin{\#_c}(\Psi h\Psi'^{-1})
=\Psi(g\mathbin{\#_c}h)\Psi'^{-1}$; in particular $H$ is replaced by
$\Psi H\Psi'^{-1}$.
\end{enumerate}
\end{lemma}

\begin{proof}
(1) With $\phi=g^{-1}h$, $h(h^{-1}g)^{1-c}=g\phi(\phi^{-1})^{1-c}=g\phi^c$.
(2) The first identity is $\mathcal Q_i=P_{\gamma_i}(P^{\GKT}_{\gamma_i})^{-1}$,
and the second follows because $H$ is an algebra automorphism.  (3) Finite
restriction commutes with composition, $\log$ and $\exp$.  After
continuation, $P_{\gamma_i}$ and $P^{\GKT}_{\gamma_i}$ become
$\Psi P_{\gamma_i}$ and $\Psi'P^{\GKT}_{\gamma_i}$, so $\mathcal Q_i$ becomes
$\Psi\mathcal Q_i\Psi'^{-1}$; moreover
$(\Psi g\Psi'^{-1})^{-1}\Psi h\Psi'^{-1}=\Psi'g^{-1}h\Psi'^{-1}$, and
rational powers commute with conjugation, since $\log$ and $\exp$ do.
\end{proof}

For $\gamma_+$ and $\gamma_-$ above, $H=Z_{-1/2}$, and
the corrected product $(Z_{1/2}\Sigma_1\Sigma_2f_1)(Z_{-1/2}\Sigma_1\Sigma_2f_2)$
is the image under $Z_{-1/2}$ of the product
$(Z_1\Sigma_1\Sigma_2f_1)(\Sigma_1\Sigma_2f_2)$ of the GKT transports.

\begin{definition}[Interpolants and corrected products]
\label{def:corrected-product}
Let each input $i$ carry a positive rational \emph{degree} $r_i$.  A
\emph{product tree} $\mathsf T$ on the inputs $1,\dots,n$ is a planar
rooted binary tree whose leaves are $1,\dots,n$ in this order, that is, a
parenthesization of a product of $n$ factors; its degree $|\mathsf T|$ is the
sum of the degrees of its leaves.  If $\mathsf T$ is the leaf $i$, its
\emph{interpolant} is $H_{\mathsf T}=\mathcal Q_i$ and its \emph{corrected
product} is $\mathsf P_{\mathsf T}=P_{\gamma_i}f_i$.  If the root of
$\mathsf T$ has left and right subtrees $\mathsf T_1,\mathsf T_2$, then
\[
 H_{\mathsf T}=H_{\mathsf T_1}\mathbin{\#_{|\mathsf T_2|/|\mathsf T|}}H_{\mathsf T_2},
 \qquad
 \mathsf P_{\mathsf T}
 =\bigl(H_{\mathsf T}H_{\mathsf T_1}^{-1}\mathsf P_{\mathsf T_1}\bigr)
  \bigl(H_{\mathsf T}H_{\mathsf T_2}^{-1}\mathsf P_{\mathsf T_2}\bigr).
\]
\end{definition}

For two leaves of equal degree, $H_{\mathsf T}=H$ and the corrections
$H_{\mathsf T}H_{\mathsf T_j}^{-1}$ are the $C_i$.  If the $\mathcal Q_i$
commute, induction on $\mathsf T$ gives
$\log H_{\mathsf T}=\sum_ir_i\log\mathcal Q_i\big/\sum_ir_i$ for every
product tree $\mathsf T$, whereas iterated midpoints, which ignore the
degrees, give three commuting residual transports the weights
$(\frac14,\frac14,\frac12)$ or $(\frac12,\frac14,\frac14)$, depending on the
parenthesization.

\begin{proposition}[Degree-weighted coherence]
\label{thm:residual-coherence}
For every product tree $\mathsf T$,
$\mathsf P_{\mathsf T}=H_{\mathsf T}\prod_iP^{\GKT}_{\gamma_i}f_i$.
For product trees $\mathsf T,\mathsf T',\mathsf T''$ on the same inputs, the
\emph{tree transition}
$\mathcal V_{\mathsf T\leftarrow\mathsf T'}=H_{\mathsf T}H_{\mathsf T'}^{-1}$
carries $\mathsf P_{\mathsf T'}$ to $\mathsf P_{\mathsf T}$, and
$\mathcal V_{\mathsf T\leftarrow\mathsf T'}
\mathcal V_{\mathsf T'\leftarrow\mathsf T''}
=\mathcal V_{\mathsf T\leftarrow\mathsf T''}$.
In particular the composite of the tree transitions around any cycle of
trees is the identity; for four inputs this includes the pentagon of the
associahedron.  Let $\mathcal U$ be the quotient of the free associative
algebra $\QQ\langle x,y,z\rangle$ by the span of the words containing some
letter twice, let $\mathcal Q_1=e^x$, $\mathcal Q_2=e^y$, $\mathcal Q_3=e^z$
in $\mathcal U$ have equal degrees, and let $\mathsf T_\ell=((1,2),3)$ and
$\mathsf T_r=(1,(2,3))$.  Then
\begin{equation}\label{eq:first-associator}
 \log\mathcal V_{\mathsf T_\ell\leftarrow\mathsf T_r}
 =\log\bigl(H_{\mathsf T_\ell}H_{\mathsf T_r}^{-1}\bigr)
 =\frac1{72}[[x,z],y].
\end{equation}
In particular $\mathcal V_{\mathsf T_\ell\leftarrow\mathsf T_r}$ is not the
identity, and the two corrected triple products differ in general.
\end{proposition}

\begin{proof}
For a leaf, $\mathsf P_{\mathsf T}=\mathcal Q_iP^{\GKT}_{\gamma_i}f_i$.  By
induction, the two corrected subtree products at the root are $H_{\mathsf T}$
applied to the products of the GKT transports over the two subtrees; since
$H_{\mathsf T}$ is an algebra automorphism, this gives
$\mathsf P_{\mathsf T}$, and the statements on tree transitions follow by
cancellation.  For $g=e^U$, $h=e^V$, the Baker--Campbell--Hausdorff formula
gives, modulo brackets of four or more factors $U,V$,
\[
 \log(g\mathbin{\#_c}h)
 =(1-c)U+cV-\tfrac1{12}c(1-c)\bigl([U,[U,V]]+[V,[V,U]]\bigr).
\]
In $\mathcal U$ every word of length at least four contains a repeated
letter and vanishes, and so do the cubic terms for $(U,V)=(x,y)$ and
$(y,z)$.  Hence $\log H_{(1,2)}=\frac12(x+y)$ and
$\log H_{(2,3)}=\frac12(y+z)$.  With $(U,V,c)=(\frac12(x+y),z,\frac13)$ and
$(x,\frac12(y+z),\frac23)$, for which $c(1-c)/12=1/54$,
\begin{align*}
 \log H_{\mathsf T_\ell}
 &=\tfrac13(x+y+z)-\tfrac1{216}\bigl([x,[y,z]]+[y,[x,z]]\bigr),\\
 \log H_{\mathsf T_r}
 &=\tfrac13(x+y+z)-\tfrac1{216}\bigl([y,[z,x]]+[z,[y,x]]\bigr).
\end{align*}
Their linear parts agree and they have no quadratic part, so their brackets
vanish in $\mathcal U$, and $\log(H_{\mathsf T_\ell}H_{\mathsf T_r}^{-1})$ is
their difference.  Since $-[y,[z,x]]=[y,[x,z]]$ and, by the Jacobi identity,
$[x,[y,z]]-[z,[y,x]]=[y,[x,z]]$, this difference is
$-\frac3{216}[y,[x,z]]=\frac1{72}[[x,z],y]$, a nonzero element of
$\mathcal U$.
\end{proof}

\subsection{Change of paths}

Let $\boldsymbol\gamma=(\gamma_i)$ and $\boldsymbol\gamma'=(\gamma'_i)$ be
tuples of paths as above such that $\gamma_i$ and $\gamma'_i$ have the same
endpoints for every $i$, and write $H_{\boldsymbol\gamma,\mathsf T}$ for the
interpolant of $\boldsymbol\gamma$ and the product tree $\mathsf T$.  The
\emph{leg maps} $\mathcal G_i^{\boldsymbol\gamma'\leftarrow\boldsymbol\gamma}
=P^{\GKT}_{\gamma'_i}(P^{\GKT}_{\gamma_i})^{-1}$ and the \emph{transitions}
$\mathcal V_{(\boldsymbol\gamma',\mathsf T')\leftarrow(\boldsymbol\gamma,\mathsf T)}
=H_{\boldsymbol\gamma',\mathsf T'}H_{\boldsymbol\gamma,\mathsf T}^{-1}$ are
automorphisms of the chart algebra of the target region.

\begin{proposition}[Change of admissible paths]
\label{prop:corridor-groupoid}
For a product tree $\mathsf T$ and $a_i=P^{\GKT}_{\gamma_i}f_i$, the
corrected product of $\boldsymbol\gamma'$ and $\mathsf T$ is
\begin{equation}\label{eq:corridor-square}
 H_{\boldsymbol\gamma',\mathsf T}\prod_iP^{\GKT}_{\gamma'_i}f_i
 =\mathcal V_{(\boldsymbol\gamma',\mathsf T)\leftarrow(\boldsymbol\gamma,\mathsf T)}
  H_{\boldsymbol\gamma,\mathsf T}
  \prod_i\mathcal G_i^{\boldsymbol\gamma'\leftarrow\boldsymbol\gamma}a_i.
\end{equation}
The leg maps satisfy
$\mathcal G_i^{\boldsymbol\gamma''\leftarrow\boldsymbol\gamma'}
\mathcal G_i^{\boldsymbol\gamma'\leftarrow\boldsymbol\gamma}
=\mathcal G_i^{\boldsymbol\gamma''\leftarrow\boldsymbol\gamma}$, and every
composite of transitions from $(\boldsymbol\gamma,\mathsf T)$ to
$(\boldsymbol\gamma',\mathsf T')$, in particular of tree transitions
($\boldsymbol\gamma'=\boldsymbol\gamma$) and of transitions changing only
the paths ($\mathsf T'=\mathsf T$), equals
$\mathcal V_{(\boldsymbol\gamma',\mathsf T')\leftarrow(\boldsymbol\gamma,\mathsf T)}$.
If $\phi\neq\mathrm{id}$ is an automorphism of the target algebra, no
automorphism carries $ab$ to $\phi(a)b$ for all $a,b$; hence, if one leg map
is not the identity and another is, the leg maps cannot be absorbed into one
automorphism of the target algebra.
\end{proposition}

\begin{proof}
The identities
$P^{\GKT}_{\gamma'_i}f_i=\mathcal G_i^{\boldsymbol\gamma'\leftarrow\boldsymbol\gamma}a_i$
and $\mathcal V_{(\boldsymbol\gamma',\mathsf T)\leftarrow(\boldsymbol\gamma,\mathsf T)}
H_{\boldsymbol\gamma,\mathsf T}=H_{\boldsymbol\gamma',\mathsf T}$ give
\eqref{eq:corridor-square}, whose left side is the corrected product by
Proposition~\ref{thm:residual-coherence}.  The leg maps and the transitions
are ratios, so their composites telescope.  Finally, an automorphism carrying
$ab$ to $\phi(a)b$ for all $a,b$ equals $\phi$, by $b=1$, and the identity,
by $a=1$.
\end{proof}
