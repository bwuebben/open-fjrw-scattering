\section{Scattering for the Jacobi bracket; hyperbolic core holonomy}
\label{sec:jacobi-theta}

\subsection{The volume line and its Jacobi bracket}

Local formulas are written in the fixed frame of the sheet $S_0\ni O_0$ of
Section~\ref{subsec:kummer-charts}, which is the frame at the root of the
path tree $\Tcal$ of Section~\ref{sec:affine-kummer}; the frames at the
other vertices of $\Tcal$ are reached by transport along the edges of the
graph $\mathfrak C$.  As in Section~\ref{subsec:exponent-lattices},
elements of $\widetilde M$, matrices acting on $\widetilde M$ and
determinants of pairs of its elements are written in $F$-coordinates, so
that $z_1=z^{(1,0)}$ and $z_2=z^{(0,1)}$ are the characters of $x$ and $y$
under the dictionary $x^ay^b\mapsto z^{F(a,b)}$.  Put
\begin{equation}\label{eq:primitive-form}
 \kappa=(1,1),
 \qquad \Omega_0=d\log z_1\wedge d\log z_2,
 \qquad \Omega_\kappa=z^\kappa\Omega_0,
\end{equation}
so that $\kappa$ is the exponent $F(1,1)$ of $xy$ and $\Omega_\kappa$
corresponds to $dx\wedge dy$.  For $q=(q_1,q_2)$ put $q^\perp=(q_2,-q_1)$,
and denote the invariant logarithmic derivation in direction $v$ by
$\partial_v$, so that $\partial_vz^m=\langle v,m\rangle z^m$.

\begin{proposition}[Shifted Hamiltonian bracket]
\label{prop:shifted-bracket}
The inverse bivector $\pi_\kappa=z^{-\kappa}\pi_0$ of $\Omega_\kappa$ has
Hamiltonian vector fields
\begin{equation}\label{eq:shifted-Hamiltonian}
 X_q^\kappa=z^{q-\kappa}\partial_{q^\perp},
\end{equation}
and
\begin{equation}\label{eq:shifted-bracket}
 [X_q^\kappa,X_r^\kappa]
 =-\det(q,r)X_{q+r-\kappa}^\kappa.
\end{equation}
Thus, after writing $q=k+\kappa$ and $r=\ell+\kappa$, the output is indexed
by $k+\ell$.
\end{proposition}

\begin{proof}
The Hamiltonian field of $z^q$ for $z^{-\kappa}\pi_0$ is
\eqref{eq:shifted-Hamiltonian}.  The bracket of two monomial derivations has
direction
\[
 \langle q^\perp,r-\kappa\rangle r^\perp
 -\langle r^\perp,q-\kappa\rangle q^\perp
 =-\det(q,r)(q+r-\kappa)^\perp,
\]
which gives \eqref{eq:shifted-bracket}.
\end{proof}

The fields $X^\kappa_q$, $q\in\widetilde M$, span over $\Bbbk$ the Lie
algebra $\mathfrak g_\kappa$ of $\pi_\kappa$-Hamiltonian derivations of
$\Bbbk[\widetilde M]$, with the bracket \eqref{eq:shifted-bracket}.  We call
$\mathfrak g_\kappa$ the \emph{Hamiltonian wall algebra}: the wall
automorphisms constructed below are exponentials of elements of
$\mathfrak g_\kappa$ with coefficients in the ideal generated by the
deformation parameters, completed with respect to that ideal; they are
written in the frame of $S_0$ and transported to the other frames by
Proposition~\ref{prop:Jacobi-transport}.  For $\kappa=0$ the field
$X^0_q=z^q\partial_{q^\perp}$ is the Hamiltonian vector field of $z^q$ for
$\Omega_0$, and $X^0_0=0$.  Thus $\mathfrak g_0$ is spanned by the fields
$z^q\partial_{q^\perp}$ with $q\neq0$, and for an Artinian local
$\Bbbk$-algebra $R$ with maximal ideal $\mathfrak m_R$ the Lie algebra
$\mathfrak g_0\otimes_\Bbbk\mathfrak m_R$ is the Lie algebra
$\mathfrak h_R$ of the tropical vertex group of
Gross--Pandharipande--Siebert \cite[Sect.~1.1]{grosspandharipandesiebert2010},
with $\widetilde M$ in place of the lattice denoted $M$ there and $\Bbbk$ in
place of $\CC$.  As in
Section~\ref{subsec:intro-notation} we write
$X_{a,b}=X^\kappa_{\kappa+(a,b)}$.

The lifted slab map $\widetilde\theta_V$ of \eqref{eq:local-root-ring} is
symplectic for $\Omega_0$, but
\begin{equation}\label{eq:slab-conformal}
 \widetilde\theta_V^*\Omega_\kappa
 =g_V^{\langle N_V,\kappa\rangle}\Omega_\kappa,
 \qquad
 \bigl(\langle N_V,\kappa\rangle\bigr)_V=(-2,-2,2,-2).
\end{equation}
By \eqref{eq:slab-conformal}, the transition law of $\Omega_\kappa$ requires
a line bundle.  Choose a local line frame $e$ and identify frames across the
slab by
\begin{equation}\label{eq:line-transition}
 \widetilde\theta_V^*e'=g_V^{-\langle N_V,\kappa\rangle}e.
\end{equation}
The products $\Omega_\kappa\otimes e$ then glue.  We call the resulting line
bundle $\Lcal_\kappa$ the \emph{volume line}.  Like the lifted slab maps
(Section~\ref{subsec:kummer-charts}), the line $\Lcal_\kappa$ and all
transports below are defined on the Kummer covers and not on the root
stacks, namely on the spectra of the rings \eqref{eq:local-root-ring} and,
over a finite rooted subtree $T\subset\Tcal$, on $\Spec R_T^{\mathrm{root}}$
of \eqref{eq:finite-kummer-ring}.

\begin{proposition}[No scalar frame]\label{prop:no-scalar-frame}
The Lie algebra of $\pi_\kappa$-Hamiltonian derivations of the structure
sheaf, which contains the two singleton fields $X^\kappa_{(2,1)}$ and
$X^\kappa_{(1,2)}$, is not preserved under conjugation by any lifted slab
transition.  Consequently its local brackets do not glue to a bracket on the
structure sheaf.
\end{proposition}

\begin{proof}
For a derivation $\delta$ let $\operatorname{div}\delta$ be defined by
$L_\delta\Omega_\kappa=(\operatorname{div}\delta)\,\Omega_\kappa$.  Every
$\pi_\kappa$-Hamiltonian derivation $X_h$ has
$\operatorname{div}X_h=0$, since $\iota_{X_h}\Omega_\kappa=\pm dh$.  If
$\theta$ is an automorphism with $\theta^*\Omega_\kappa=G\Omega_\kappa$, then
$L_{\theta^{-1}\delta\theta}=\theta^{-1}L_\delta\theta$ on forms gives
\[
 \operatorname{div}\bigl(\theta^{-1}\delta\theta\bigr)
 =\theta^{-1}\Bigl(\operatorname{div}\delta+\frac{\delta(G)}{G}\Bigr).
\]

Let $u_V$, $\epsilon_V$ and $x_V=z^{2\epsilon_Vu_V}$ be as in
\eqref{eq:root-data}, so that $\langle N_V,u_V\rangle=0$ and the pulled-back
slab function $f_V$ is a nonconstant polynomial in $x_V$.  Then
$\widetilde\theta_V$ fixes $f_V$ and $g_V$.  Take $\theta=\widetilde\theta_V$,
$G=g_V^{\langle N_V,\kappa\rangle}$, and $\delta=X_q^\kappa$ with
$q\in\{(2,1),(1,2)\}$.  Since $g_V^{d_V}=f_V$,
\[
 \operatorname{div}\bigl(\widetilde\theta_V^{-1}X_q^\kappa\widetilde\theta_V\bigr)
 =\frac{\langle N_V,\kappa\rangle}{d_V}\,
  \widetilde\theta_V^{-1}\!\left(\frac{X_q^\kappa(f_V)}{f_V}\right),
 \qquad
 X_q^\kappa(f_V)=2\epsilon_V\langle q^\perp,u_V\rangle\,z^{q-\kappa}x_Vf_V'(x_V),
\]
where $f_V'$ is the derivative of $f_V$ as a polynomial in $x_V$.  The
pairings $\langle q^\perp,u_V\rangle$ are
\[
\begin{array}{c|rrrr}
  &\widetilde A&B_0&B_1&\widetilde C\\ \hline
 q=(2,1)&1&3&-1&-5\\
 q=(1,2)&5&3&7&-1,
\end{array}
\]
all nonzero, and $\langle N_V,\kappa\rangle=\pm2$.  Hence every conjugate of
a singleton field by a lifted slab transition has nonzero divergence and is
not $\pi_\kappa$-Hamiltonian.  The argument uses only that $f_V$ is a
nonconstant polynomial in $x_V$, so it applies at every value of the
slab parameter, including the Fermat value $a=0$.  If the local brackets
glued to a bracket on the structure sheaf, each $\widetilde\theta_V$ would
be a Poisson map for $\pi_\kappa$, and conjugation by it would preserve
Hamiltonian derivations.
\end{proof}

By Proposition~\ref{prop:no-scalar-frame} the bracket of $\pi_\kappa$ cannot
be glued on functions; Proposition~\ref{prop:Jacobi-transport} glues it as a
bracket on the sections of $\Lcal_\kappa$.  Recall that a \emph{Jacobi
bracket}, or Jacobi structure, on a line bundle $L$ is a Lie bracket on
local sections of $L$ that is a first-order bidifferential operator
\cite{kirillov1976,lichnerowicz1978,vitaglianowade2016} (a local Lie algebra
with one-dimensional fibers in the sense of Kirillov; the notion is unrelated
to the Jacobian algebra $\Jac(W)$ of a potential).  In a local frame $e$ of
$L$ it is given by a bivector $\Lambda$ and a vector field $E$ through
\begin{equation}\label{eq:Jacobi-local}
 \{fe,he\}=\bigl(\Lambda(df,dh)+fE(h)-hE(f)\bigr)e.
\end{equation}
For $E=0$ the coefficient of $e$ is the Poisson bracket $\Lambda(df,dh)$.

\begin{proposition}[Jacobi transport]\label{prop:Jacobi-transport}
The line $\Lcal_\kappa$ over the path tree $\Tcal$ carries a Jacobi bracket,
determined by the choice of the frame of $S_0$ at the root of $\Tcal$, whose
expression in that frame is the Poisson bracket of $\pi_\kappa$.  Every
lifted slab transition is a Jacobi isomorphism.  The construction is
compatible with finite root restriction and equivariant under the deck
action of $\Pi$.
\end{proposition}

\begin{proof}
In a frame $e$, write a Jacobi bracket as in \eqref{eq:Jacobi-local}.
Under $e'=ae$ the local pair transforms by
\begin{equation}\label{eq:Jacobi-frame-change}
 \Lambda'=a\Lambda,
 \qquad E'=aE+\Lambda^\sharp(da).
\end{equation}
Start at the root of $\Tcal$, in the frame of $S_0$, with
$(\Lambda,E)=(\pi_\kappa,0)$ and transport the pair across each edge using
\eqref{eq:line-transition} and \eqref{eq:Jacobi-frame-change}.  Every vertex
of the path tree has a unique route from the root, so the construction is
unambiguous and compatible with restriction to finite subtrees.  Change of
frame and pullback preserve the Jacobi identity.  The deck action transports
the same data, giving deck equivariance.
\end{proof}

The bracket depends on the choice of base frame.  At each singular point
$V$ the multiplier $a_V=g_V^{-\langle N_V,\kappa\rangle}=g_V^{\pm2}$ in
\eqref{eq:line-transition} is nonconstant and $\pi_\kappa$ is
nondegenerate, so by \eqref{eq:Jacobi-frame-change} with $E=0$ and $a=a_V$
the pair transported across the arc carrying $\widetilde\theta_V$ has a
vector field which, expressed in the chart before the arc, is
$\pi_\kappa^\sharp(da_V)\neq0$.  Starting instead from
$(\pi_\kappa,0)$ in the frame beyond that arc therefore gives a different
Jacobi bracket.

The Jacobi bracket of $\Lcal_\kappa$ can equivalently be described by its
Poissonization, a homogeneous Poisson structure on
$\Lcal_\kappa^\vee\setminus0$ \cite{vitaglianowade2016}.  We use the
line-bundle formulation because it displays the slab multiplier in
\eqref{eq:line-transition} explicitly.

\begin{definition}[Jacobi walls and consistency]\label{def:Jacobi-wall}
Fix a finite rooted subtree $T\subset\Tcal$ and work on the Kummer covers
over $T$ modulo $\mathfrak m^N$, where $\mathfrak m=(t_1,t_2)$ is the
singleton ideal of Definition~\ref{def:pathwise-root-object}.  A
\emph{Jacobi wall} is a pair $(\mathfrak d,\Theta_{\mathfrak d})$, where
$\mathfrak d$, the \emph{support} of the wall, is a cooriented rational
affine segment, ray, or full chord (a segment both of whose ends lie on the
boundary) in the smooth locus $\Acal^\circ$, and $\Theta_{\mathfrak d}$ is a
locally constant unipotent Jacobi automorphism along $\mathfrak d$,
congruent to the identity modulo $\mathfrak m$.  The automorphism is written
in the local frame of $\Lcal_\kappa$ and is transported between charts by
Proposition~\ref{prop:Jacobi-transport}; the charts and the transitions
between them are placed as in Section~\ref{subsec:kummer-charts}.

Let $\gamma$ be a path transverse to the walls.  A crossing is
\emph{positive} if the velocity of $\gamma$ pairs positively with the
coorientation of the wall; a positive crossing contributes
$\Theta_{\mathfrak d}$ and a negative crossing contributes
$\Theta_{\mathfrak d}^{-1}$.  The \emph{path-ordered product}
$\Theta_\gamma$ is the composite of these contributions, each transported to
the frame at the initial point of $\gamma$ along $\gamma$, with the first
crossing acting first; in a product of operators the rightmost factor acts
first.  A \emph{joint} of a locally finite collection of Jacobi walls is a
point of $\Acal^\circ$ at which two supports of different directions meet or
at which a support ends, and the collection is \emph{consistent at a joint}
if the path-ordered product around a sufficiently small positively oriented
loop about that joint is the identity.  The \emph{regions} of the collection
are the connected components of the complement in $\Acal^\circ$ of the
supports of those walls whose automorphisms are not the identity; for the
wall structure $\mathfrak D$ of Theorem~\ref{thm:annular-consistency} we
speak of the \emph{regions of $\mathfrak D$}.
\end{definition}

All these notions are finite-stage notions.  A wall structure over $\Tcal$
means a compatible system of such structures as $T$ and $N$ vary, and its
path-ordered products are the compatible systems of the finite-stage
path-ordered products.

\subsection{Scattering of the two initial Hamiltonians}

With the singleton parameters $t_1,t_2$, put
\begin{equation}\label{eq:singleton-generators}
 E_{a,b}=t_1^at_2^bX_{a,b}=t_1^at_2^bX_{\kappa+(a,b)}^\kappa,
 \qquad (a,b)\in\NN^2\setminus\{(0,0)\}.
\end{equation}
The \emph{shifted exponent} of $E_{a,b}$ is $\kappa+(a,b)=(a+1,b+1)$, and its
\emph{shifted slope} is $(b+1)/(a+1)$.  By \eqref{eq:shifted-bracket},
\begin{equation}\label{eq:singleton-bracket}
 [E_{a,b},E_{c,d}]
 =-\det\bigl(\kappa+(a,b),\kappa+(c,d)\bigr)E_{a+c,b+d}.
\end{equation}
The two incoming GKT normalizations are
\begin{equation}\label{eq:incoming-singletons}
 \mathsf A=-\frac12E_{1,0},
 \qquad \mathsf B=-\frac12E_{0,1}.
\end{equation}
Since the shifted exponents of $\mathsf A$ and $\mathsf B$ are $(2,1)$ and
$(1,2)$,
\begin{equation}\label{eq:first-bracket}
 [\mathsf A,\mathsf B]= -\frac34E_{1,1}.
\end{equation}

Let $\widehat{\mathfrak g}_{\mathrm{sing}}$ be the product of the lines
spanned by the $E_{a,b}$, completed in the total degree $a+b$; it is a Lie
subalgebra of the $\mathfrak m$-adic completion of $\mathfrak g_\kappa\otimes_\Bbbk
\Bbbk[t_1,t_2]$.  For a ray
$\mathfrak r\subset\RR_{\geq0}^2$ we let $\widehat{\mathfrak g}_{\mathfrak r}$
be the closed span of the $E_{a,b}$ whose shifted exponent lies on
$\mathfrak r$.  By \eqref{eq:singleton-bracket} each
$\widehat{\mathfrak g}_{\mathfrak r}$ is abelian, and we call the abelian
group $\exp(\widehat{\mathfrak g}_{\mathfrak r})$ the \emph{ray group} of
$\mathfrak r$.  Rays are ordered by their slopes, and $\mathfrak r_{2,1}$,
$\mathfrak r_{1,2}$ denote the rays of $(2,1)$ and $(1,2)$, the
\emph{incoming rays}.

\begin{proposition}[Ordered singleton completion]
\label{prop:singleton-completion}
Let operators act on functions, the rightmost factor acting first.
\begin{enumerate}
\item There are unique elements
$C_{\mathfrak r}\in\widehat{\mathfrak g}_{\mathfrak r}$ such that
\begin{equation}\label{eq:singleton-commutator}
 e^{\mathsf A}e^{-\mathsf B}e^{-\mathsf A}e^{\mathsf B}
 =\overrightarrow{\prod_{\mathfrak r}}\exp(C_{\mathfrak r}),
\end{equation}
the product taken over rays with slopes increasing from left to right.
Modulo $\mathfrak m^N$ only finitely many $C_{\mathfrak r}$ are nonzero, and the
factorization is compatible with reduction of $N$.
\item Write $C_{\mathfrak r}=\sum c_{a,b}E_{a,b}$.  Then $c_{a,b}=0$ unless
$a,b\geq1$ and $(a+1,b+1)$ lies in the cone spanned by $(2,1)$ and $(1,2)$.
Moreover $c_{b,a}=c_{a,b}$, $c_{1,1}=\frac34$, and the coefficients of total
degree at most $10$ are those of Table~\ref{tab:singleton-coefficients}.
\item Equivalently,
\begin{equation}\label{eq:singleton-increasing-slope}
 e^{\mathsf A}e^{\mathsf B}
 =e^{\mathsf B-C_{\mathfrak r_{1,2}}}
  \Bigl(\overleftarrow{\prod_{\frac12<\mathrm{slope}(\mathfrak r)<2}}
  e^{-C_{\mathfrak r}}\Bigr)
  e^{\mathsf A-C_{\mathfrak r_{2,1}}},
\end{equation}
where the middle product has slopes decreasing from left to right.  This is
the unique factorization of $e^{\mathsf A}e^{\mathsf B}$ into ray groups in
which the factor of least slope acts first.
\end{enumerate}
\end{proposition}

\begin{proof}
The bracket \eqref{eq:singleton-bracket} is additive in $(a,b)$.  Hence the
elements of total degree at least $d$ form an ideal
$\widehat{\mathfrak g}^{\geq d}$, the quotient
$\widehat{\mathfrak g}^{\geq d}/\widehat{\mathfrak g}^{\geq d+1}$ is central
in $\widehat{\mathfrak g}_{\mathrm{sing}}/\widehat{\mathfrak g}^{\geq d+1}$,
and modulo $\mathfrak m^N$ only degrees below $N$ survive.  Let
$\mathsf W=e^{\mathsf A}e^{-\mathsf B}e^{-\mathsf A}e^{\mathsf B}$ be the
left side of \eqref{eq:singleton-commutator}, and suppose that a product $P$
of ray factors satisfies $\mathsf WP^{-1}\in\exp(\widehat{\mathfrak g}^{\geq d})$.  The
degree-$d$ part of $\log(\mathsf WP^{-1})$ decomposes uniquely into ray components;
since this part is central modulo degree $d+1$, adding each component to the
logarithm of the corresponding factor removes the discrepancy in degree $d$,
and no other choice does.  Induction on $d$ gives existence and uniqueness
in (1).  This is the argument of the Kontsevich--Soibelman lemma
\cite[Thm.~1.4]{grosspandharipandesiebert2010}; it uses only the grading and
not convexity, which fails here: the shifted exponent $q+r-\kappa$ of a
bracket need not lie between $q$ and $r$.

For the support statement, let $\mathcal C$ be the cone spanned by $(2,1)$
and $(1,2)$.  If $q,r\in\mathcal C\cap\ZZ^2$ are not parallel, then
\[
 2(q_2+r_2-1)-(q_1+r_1-1)=(2q_2-q_1)+(2r_2-r_1)-1\geq0,
\]
because the two nonnegative integers $2q_2-q_1$ and $2r_2-r_1$ vanish
together only when $q$ and $r$ both lie on the ray of $(2,1)$; the other
inequality is symmetric.  Parallel shifted exponents have vanishing bracket.
Hence the closed span of the $E_{a,b}$ with shifted exponent in $\mathcal C$
is a Lie subalgebra.  It contains $\mathsf A$ and $\mathsf B$, and the
inductive construction stays in it.  Its only generators with $a=0$ or $b=0$
are $E_{1,0}$ and $E_{0,1}$, of degree one, and the left side of
\eqref{eq:singleton-commutator} is the identity modulo degree two.  This
proves the support statement.  Modulo degree three the left side equals
$\exp(-[\mathsf A,\mathsf B])=\exp(\tfrac34E_{1,1})$ by
\eqref{eq:first-bracket}.

For the symmetry, the linear map $\psi(E_{a,b})=-E_{b,a}$ preserves
\eqref{eq:singleton-bracket}, since exchanging the coordinates of both
vectors changes the sign of the determinant.  It sends $\mathsf A$ to
$-\mathsf B$ and $\mathsf B$ to $-\mathsf A$, hence $\mathsf W$ to
$e^{-\mathsf B}e^{\mathsf A}e^{\mathsf B}e^{-\mathsf A}=\mathsf W^{-1}$, and it
reverses the order of slopes.  Applying $\psi$ to
\eqref{eq:singleton-commutator}, and inverting
\eqref{eq:singleton-commutator}, gives two factorizations of $\mathsf W^{-1}$ with
slopes decreasing from left to right.  By uniqueness,
$\psi(C_{\mathfrak r})=-C_{\mathfrak r'}$, where $\mathfrak r'$ is the
reflection of $\mathfrak r$; comparing coefficients gives $c_{b,a}=c_{a,b}$.

For (3), invert \eqref{eq:singleton-commutator}, multiply by
$e^{\mathsf B}$ on the left and by $e^{\mathsf A}$ on the right, and use that
$C_{\mathfrak r_{1,2}}$ commutes with $\mathsf B$ and $C_{\mathfrak r_{2,1}}$
with $\mathsf A$.  The factorization with the least slope acting first is
unique by the same induction.  The entries of
Table~\ref{tab:singleton-coefficients} are computed by solving the
degree-by-degree recursion in the proof of (1) through total degree ten.
\end{proof}

All coefficients in Table~\ref{tab:singleton-coefficients} are positive, and
every bidegree of total degree at most $10$ allowed by (2) occurs.  The
factors $\exp(C_{\mathfrak r_{2,1}})$ and $\exp(C_{\mathfrak r_{1,2}})$ on
the incoming rays are nontrivial from total degree four on: their first
coefficients are $c_{3,1}=c_{1,3}=\frac1{16}$.

\subsection{Wall supports and consistency at the joints}
\label{subsec:cofactor-supports}

We work in the chart of $\Delta$ drawn in
Figure~\ref{fig:cofactor-supports}(a), with affine coordinates $(s_1,s_2)$
in which
$\Delta=\operatorname{conv}\{(-1,-1),(-1,1),(3,-1)\}$ and the central vertex
$O$ of the star subdivision is the origin.  For $V\in\{A,B,C\}$ let $p_V$ be
the vertex of $\Delta$ on the star edge through $V$, so that
$p_A=(-1,-1)$, $p_B=(-1,1)$, $p_C=(3,-1)$.  In the normalization of
Appendix~\ref{app:calculations}, $V$ is the midpoint of the edge from $O$ to
$p_V$, so that $A=(-\frac12,-\frac12)$, $B=(-\frac12,\frac12)$ and
$C=(\frac32,-\frac12)$.  Points and directions in this chart are given by
their coordinates $(s_1,s_2)$, and linear forms by their coefficients with
respect to $s_1,s_2$; shifted exponents, which are elements of
$\widetilde M$, are written in $F$-coordinates.

\emph{Transitions and frames.}  The transitions at the singular points,
including the line transitions \eqref{eq:line-transition}, are placed as in
Section~\ref{subsec:kummer-charts}, on the cuts over the outer half-edges
$c_V$ from $V$ to $p_V$, and each sheet $S_i\ni O_i$ carries one frame.  By
Proposition~\ref{prop:proper-supports} below, no support meets an arc
carrying a transition and the supports through $O_i$ lie in $S_i$; hence the
factor on every support through $O_i$ is written in the frame of $S_i$ along
the whole support.

\emph{Supports.}  The incoming shifted exponents $(2,1)$ and $(1,2)$ have
the support directions, in the chart $(s_1,s_2)$,
\begin{equation}\label{eq:cofactor-singletons}
 L(2,1)=(-4,3),
 \qquad L(1,2)=(-2,3).
\end{equation}
Each is placed on the full chord through $O$ in its direction.  For
$q\in\{(2,1),(1,2)\}$ the \emph{positive} and \emph{negative half-chords} of
$Lq$ are the parts of this chord in $\RR_{\geq0}Lq$ and in $\RR_{\leq0}Lq$.
Full chords are forced: if a degree-one wall ended at an interior point, a
small loop around that endpoint would cross the wall once, while every
correction begins in degree two.  If $q=c_1(2,1)+c_2(1,2)$ with
$c_1,c_2\geq0$, then, in the chart $(s_1,s_2)$,
\begin{equation}\label{eq:outgoing-cofactor}
 Lq=(-4c_1-2c_2,3c_1+3c_2).
\end{equation}
For $c_1>c_2$ the ray $\RR_{\geq0}Lq$ leaves $\Delta$ through the facet
$s_1=-1$, and for $c_2>c_1$ through the facet $s_1+2s_2=1$; these are
the two facets meeting at $p_B$.  For $c_1=c_2$ it runs along the inner
half-edge from $O$ towards $p_B$ and reaches $B$.  For each ray
$\mathfrak r$ with $\frac12<\mathrm{slope}(\mathfrak r)<2$, the
\emph{outgoing wall} of $\mathfrak r$ is the ray $\RR_{\geq0}Lq$,
$q\in\mathfrak r$; for the diagonal ray it is the segment from $O$ to $B$.
The factors of the incoming rays $\mathfrak r_{2,1}$ and $\mathfrak r_{1,2}$
are placed on the positive half-chords of $L(2,1)$ and $L(1,2)$.

\emph{Placement.}  The wall of the Hamiltonian with shifted exponent $q$ is
placed along $Lq$, the image of its exponent under the gradient map of an
auxiliary quadratic function, rather than along the annihilator of the
exponent, which is what the orthogonality of dual cells in the discrete
Legendre transform would suggest.  Section~\ref{subsec:exponent-lattices}
explains the choice: among linear rules, the exchange of $x$ and $y$ restricts
the placement to a two-parameter family, of which the cofactor is one member.
Consistency at the joints uses only the cyclic order of these directions
about $O$.  Since $L$ reverses orientation ($\det L=-2$ in these coordinates;
$F$ reverses orientation and $d\psi$ preserves it) and maps the cone of shifted exponents into the half-plane $s_2>0$, a path
passing above $O$ from left to right meets the supports in increasing shifted
slope.

\emph{Coorientations.}  We coorient every wall so that crossing it in the
direction of increasing shifted slope is the positive crossing.  Thus the
chord and the outgoing wall in the direction $Lq$,
$q=c_1(2,1)+c_2(1,2)$, are cooriented, each chord along its whole length,
by the linear form
\[
 (3c_1+3c_2)\,s_1+(4c_1+2c_2)\,s_2,
\]
which vanishes on $Lq$.  We call its coefficient vector
$(3c_1+3c_2,4c_1+2c_2)$ the \emph{normal} of the wall; by
\eqref{eq:cofactor-identity} it equals $(Lq)^\perp=\iota(q^\perp)$, the
image under $\iota$ of the direction $q^\perp$ of the derivation
$X^\kappa_q$.  In the chart $(s_1,s_2)$, the normals of the two chords are
$(3,4)$ and $(3,2)$, and that of the diagonal wall is proportional to
$(1,1)$.  Every normal has positive first coordinate, so a path moving in
the direction of increasing $s_1$ crosses every wall through $O$ positively.
With $C_{\mathfrak r}$ the elements of
Proposition~\ref{prop:singleton-completion}, whose coefficients $c_{a,b}$ are
listed in Table~\ref{tab:singleton-coefficients}, the walls carry the
following factors.
\begin{enumerate}
\item The negative half-chords of $L(2,1)$ and $L(1,2)$ carry $e^{\mathsf A}$
and $e^{\mathsf B}$.
\item The positive half-chords carry
$e^{\mathsf A}\exp(-C_{\mathfrak r_{2,1}})=e^{\mathsf A-C_{\mathfrak r_{2,1}}}$
and $e^{\mathsf B}\exp(-C_{\mathfrak r_{1,2}})=e^{\mathsf B-C_{\mathfrak r_{1,2}}}$.
\item The outgoing wall of $\mathfrak r$ carries $\exp(-C_{\mathfrak r})$.
In particular the diagonal wall, the outgoing wall of the diagonal ray
$\mathfrak r_{1,1}$, carries $\exp(-C_{\mathfrak r_{1,1}})$, which is
$\exp(-\frac34E_{1,1})$ modulo $\mathfrak m^4$; it is the first outgoing wall, the
only one whose factor is nontrivial modulo $\mathfrak m^3$.
\end{enumerate}
These are the coorientations and factors of the table in
Section~\ref{sec:positive-descendents}.

\begin{proposition}\label{prop:proper-supports}
Modulo $\mathfrak m^N$, the supports placed above along the cofactor
directions, lifted to $\Acal^\circ$, form a locally finite collection of
properly embedded arcs in $\Acal^\circ$.  Its only joints are the two lifts
$O_0,O_1$ of $O$.  No support meets an arc carrying a transition, the
supports through $O_i$ lie in the sheet $S_i$, and the diagonal supports end at $B_0$
and $B_1$.  The lifts of the supports to the universal cover of
$\Acal^\circ$ meet only at lifts of $O_0$ and $O_1$, even where their
images under the developing map intersect in an affine plane.
\end{proposition}

\begin{proof}
Modulo $\mathfrak m^N$ there are finitely many deformation bidegrees.  Since the
factors $C_{\mathfrak r_{2,1}},C_{\mathfrak r_{1,2}}$ on the incoming rays
(from total degree four) are placed on the positive half-chords, the compact
polygon contains a finite embedded chord-and-ray skeleton whose members meet
only at $O$.  In the chart $(s_1,s_2)$, the positive half-chords leave
$\Delta$ at $(-1,\frac34)$ and $(-\frac12,\frac34)$.  The negative
half-chords cross the branch segment $s_2=-\frac12$ at $(\frac23,-\frac12)$ and $(\frac13,-\frac12)$, away from
$O$, and leave $\Delta$ at $(\frac43,-1)$ and $(\frac23,-1)$.  The
off-diagonal rays lie in $s_2>0$ and leave $\Delta$ through the two facets
at $p_B$.  The segments $c_A$ and $c_C$ lie in $s_2<0$, with $s_1\leq-\frac12$
and $s_1\geq\frac32$ respectively, and $c_B$ lies on the line $s_1+s_2=0$
beyond $B$, which the other supports meet only at $O$.  Hence no support
meets $c_A\cup c_B\cup c_C$, and the diagonal support ends at $B$.  Crosswise
gluing lifts each chord smoothly and produces two copies of the skeleton, one
in each sheet, without creating another intersection; the arcs carrying the
transitions lie over $c_A\cup c_B\cup c_C$, so no support meets them.
Deleting the endpoints on the boundary and at $B_0,B_1$ makes each support
relatively closed.  A compact subset of $\Acal^\circ$ stays a positive distance from
these ends and therefore meets finitely many supports.  Finally, each
support is an embedded arc in $\Acal^\circ$, so its lifts to the universal
cover are pairwise disjoint, and lifts of two different supports can meet
only over a point at which the supports meet, that is, over $O_0$ or $O_1$.
\end{proof}

Place the factors listed under \emph{Coorientations} on the
supports through $O_0$, in the frame of $S_0$, and transport them by
Proposition~\ref{prop:Jacobi-transport}: at a vertex of the path tree over
the sheet $S_i$, the supports through $O_i$ carry the factors transported
along the path of that vertex.

\begin{theorem}[Consistency at the joints]\label{thm:annular-consistency}
The transported factors on the supports of
Proposition~\ref{prop:proper-supports} define a deck-equivariant Jacobi wall
structure $\mathfrak D$ on the Kummer covers over the path tree $\Tcal$.  It is
consistent at $O_0$ and $O_1$.  For every $N$, the path-ordered product of a
compact transverse path modulo $\mathfrak m^N$ is defined on the Kummer covers over
one finite rooted subtree $T\subset\Tcal$ and depends only on the homotopy
class of the path in $\Acal^\circ$ relative to its endpoints.
\end{theorem}

\begin{proof}
By Proposition~\ref{prop:proper-supports} and the placement of the
transitions, a small disk about $O_0$ meets only the chords and outgoing
walls through $O_0$ and lies in one chart and one frame.  Let $\gamma_+$
and $\gamma_-$ run from $(s_1,s_2)=(-\varepsilon,0)$ to
$(s_1,s_2)=(\varepsilon,0)$ along $s_2=\varepsilon/10$ and
$s_2=-\varepsilon/10$, joined to their endpoints by
vertical segments; these are the paths of
Section~\ref{sec:positive-descendents}.  By \eqref{eq:outgoing-cofactor} a
support of direction $Lq$ meets $s_2=\varepsilon/10$ at
$s_1=-\frac{\varepsilon}{10}\,\frac{4c_1+2c_2}{3c_1+3c_2}$, which
increases with the slope of $q$.  Hence $\gamma_+$ meets the positive
$(2,1)$-half-chord, then the outgoing walls in increasing shifted slope, then the
positive $(1,2)$-half-chord.  The path $\gamma_-$ meets only the two negative
half-chords, first that of $(1,2)$ and then that of $(2,1)$.  Both paths
move in the direction of increasing $s_1$ where they meet a wall, so they
cross every wall positively.  The vertical segments meet no support.  With
the factors listed under \emph{Coorientations}, and the first crossing
acting first,
\[
 \Theta_{\gamma_-}=e^{\mathsf A}e^{\mathsf B},
 \qquad
 \Theta_{\gamma_+}
 =e^{\mathsf B-C_{\mathfrak r_{1,2}}}
  \Bigl(\overleftarrow{\prod_{\frac12<\mathrm{slope}(\mathfrak r)<2}}
  e^{-C_{\mathfrak r}}\Bigr)
  e^{\mathsf A-C_{\mathfrak r_{2,1}}}.
\]
Since $C_{\mathfrak r_{1,2}}$ commutes with $\mathsf B$ and
$C_{\mathfrak r_{2,1}}$ with $\mathsf A$,
\[
 \Theta_{\gamma_+}
 =e^{\mathsf B}\Bigl(\overrightarrow{\prod_{\mathfrak r}}
 \exp(C_{\mathfrak r})\Bigr)^{-1}e^{\mathsf A}
 =e^{\mathsf B}\bigl(e^{-\mathsf B}e^{\mathsf A}e^{\mathsf B}e^{-\mathsf A}\bigr)
 e^{\mathsf A}
 =e^{\mathsf A}e^{\mathsf B}=\Theta_{\gamma_-}
\]
by \eqref{eq:singleton-commutator}; equivalently, $\Theta_{\gamma_+}=\Theta_{\gamma_-}$
is \eqref{eq:singleton-increasing-slope}.  The loop formed by $\gamma_+$
and the reverse of $\gamma_-$ encircles $O_0$ once, so the path-ordered
product around a small loop about $O_0$ is the identity modulo $\mathfrak m^N$ for
every $N$.

The deck involution of the parity cover $p$ preserves orientation and is
the identity in the charts pulled back from $\Delta$.  It carries $O_0$ and
the supports through $O_0$ onto $O_1$ and the supports through $O_1$,
together with their crossing order and coorientations.  The factors on the
supports through $O_1$ are the transports of those through $O_0$ by one
Jacobi isomorphism, so the product around $O_1$ is the transport of the
identity.  The same argument applies at
every vertex of the path tree.  Transport along the path tree is by Jacobi
isomorphisms (Proposition~\ref{prop:Jacobi-transport}), which gives deck
equivariance.

A compact path crosses finitely many walls modulo $\mathfrak m^N$.  Theorem
\ref{thm:forced-category} places all of its root and wall transitions over
one finite rooted subtree.  A homotopy in $\Acal^\circ$ relative to the
endpoints can be decomposed into moves across a single wall, which do not
change $\Theta_\gamma$ because the factors are locally constant along the walls;
moves across $O_0$ or $O_1$, which do not change it by consistency; and
moves across an arc carrying a transition, which do not change it because
transport of frames is invariant under homotopy in $\Acal^\circ$ and no wall
meets such an arc.  This gives homotopy invariance.
\end{proof}

We call $\mathfrak D$ the \emph{diagram}.  Its consistency is the local
Kontsevich--Soibelman condition for walls in one chart at the two joints,
together with homotopy invariance in $\Acal^\circ$; nothing is imposed around a singular point or around the
core, where the prescribed affine, root and line holonomy is not
trivialized.  The diagonal walls end at $B_0$ and $B_1$: the terms
$c_{a,a}E_{a,a}$ of $C_{\mathfrak r_{1,1}}$ have exponents $(a,a)$, which
the monodromy at $B_0$ does not fix, since $H_{B,0}(1,1)=(3,-1)$, and
likewise at $B_1$ after transport.  A small loop about $B_i$ crosses the arc
carrying the transition at $B_i$ and the diagonal wall once each, so its
transport is the prescribed transition at $B_i$ composed with the diagonal
wall factor $\exp\bigl(-C_{\mathfrak r_{1,1}}\bigr)$ or its inverse.

\begin{definition}[Admissible paths]\label{def:framed-path}
An \emph{admissible path} $\mathfrak p$ is a reduced edge path in the graph
$\mathfrak C$ beginning at the sheet $S_0$, that is, a vertex of the path
tree $\Tcal$, together with a finite rooted subtree $T\subset\Tcal$ and a
deformation order $N$ such that all transition maps along the path, modulo
$\mathfrak m^N$, are defined over
$R_T^{\mathrm{root}}\otimes_\Bbbk R/\mathfrak m^N$, that is, over $U_T$ on
the Kummer cover $P_T$ of the finite root-stack presentation of $T$
(Theorem~\ref{thm:forced-category}).
\end{definition}

\subsection{Hyperbolic holonomy and finite covers}
\label{subsec:core-holonomy}

The matrix $K$ in \eqref{eq:intro-K} satisfies
\begin{equation}\label{eq:K-spectrum}
 \chi_K(\theta)=\det(\theta\,\mathrm{Id}-K)=\theta^2-18\theta+1,
 \qquad \theta_\pm=9\pm4\sqrt5.
\end{equation}
For $q_1=(2,1)$ and $q_2=(1,2)$, put $q_{i,n}=K^nq_i$.  Then
\begin{equation}\label{eq:Pell-recurrence}
 q_{i,n+2}=18q_{i,n+1}-q_{i,n},
 \qquad \det(q_{1,n},q_{2,n})=3.
\end{equation}

If a path $\gamma$ has linear holonomy $H_\gamma$, the transport of the
coefficient system along $\gamma$, by slab transitions and chart changes and
without wall factors, has in a developed chart the form
\begin{equation}\label{eq:coefficient-holonomy}
 \Psi_\gamma(z^m)=c_\gamma(m)z^{H_\gamma m},
\end{equation}
where $c_\gamma$ is a multiplicative character with values in the group of
units generated by the transported roots.  Each slab transition preserves
$\Omega_0$, and so does each chart change, whose linear part has determinant
one.  Hence
$\Psi_\gamma(\Omega_\kappa)=c_\gamma(\kappa)z^{H_\gamma\kappa-\kappa}\Omega_\kappa$,
and the compatible transport of $\Lcal_\kappa$ has multiplier
\begin{equation}\label{eq:line-holonomy}
 c_\gamma(\kappa)^{-1}z^{\kappa-H_\gamma\kappa}.
\end{equation}
Thus the annular holonomy is a semidirect action on exponents, roots, and the
volume line.

\begin{theorem}[Hyperbolic core holonomy]\label{thm:theta-local-system}
Let $\gamma_{\mathrm{core}}=\gamma_C\gamma_A^{-1}$ be the core loop of
Section~\ref{subsec:parity-cover}, which separates $\{\widetilde A,B_0\}$
from $\{B_1,\widetilde C\}$ and has linear holonomy $K$ on
$\widetilde M$.
\begin{enumerate}
\item $K$ acts trivially on the fractional-exponent group
$\widetilde M/M$ and hyperbolically on $\widetilde M$; its
eigenvalues \eqref{eq:K-spectrum} are not roots of unity.
\item No nonzero $m\in\widetilde M$ has finite $K$-orbit.  Consequently the
local system $\widetilde M$ on $\Acal^\circ$ has no nonzero global section,
and neither does its pullback to any connected finite cover of
$\Acal^\circ$.
\item Let $\Psi=\Psi_{\gamma_{\mathrm{core}}}$.  For every nonzero
$m\in\widetilde M$ the elements $\Psi^n(z^m)$, $n\in\ZZ$, are units with
pairwise distinct exponents $K^nm$; hence the $\Psi$-orbit of $z^m$ is
infinite, and the sum of its elements does not converge $\mathfrak m$-adically.
\end{enumerate}
\end{theorem}

\begin{proof}
Appendix~\ref{app:calculations} shows that $K$ acts trivially on
$\widetilde M/M$ and has characteristic polynomial $\chi_K$.  If $K^rm=m$
with $r\neq0$, then $m=0$, because
$\det(K^r-\mathrm{Id})=2-\theta_+^r-\theta_+^{-r}$ is nonzero.  A connected
cover of finite degree $d$ contains a closed lift of
$\gamma_{\mathrm{core}}^r$ for some $1\leq r\leq d$, and a global section of
the pulled-back local system is fixed by $K^r$.  This proves (2).  For (3),
the transport along $\gamma_{\mathrm{core}}$ carries no slab factor
(Section~\ref{subsec:kummer-charts}), so $c_{\gamma_{\mathrm{core}}}=1$ in
\eqref{eq:coefficient-holonomy} and $\Psi^n(z^m)=z^{K^nm}$ is a unit, hence
not in $\mathfrak m$.  By (2) the exponents $K^nm$ are pairwise distinct for $m\neq0$,
so the orbit is infinite.  An infinite sum converges $\mathfrak m$-adically only if
all but finitely many of its terms lie in $\mathfrak m$.
\end{proof}

Exponents therefore cannot serve as global labels: a labelling of sections
by one fixed copy of $\widetilde M$, compatible with parallel transport of
exponents, would be a trivialization of the local system $\widetilde M$ over
$\Acal^\circ$, and by Theorem~\ref{thm:theta-local-system}(2) there is none,
even after passing to a finite cover.  Nor can single-valued elements be
obtained from monomials by averaging over the core: by
Theorem~\ref{thm:theta-local-system}(3) the $\Psi$-orbit of every
nonconstant monomial is infinite, and its sum diverges.  For these reasons
we do not define broken lines or theta functions on $\Acal^\circ$.

Two finite covers retain the finite grading data.  The first is the
fractional-exponent cover $p_{\mathrm{fr}}$ of
Section~\ref{subsec:kummer-charts}, of degree two, with deck involution
$\delta_{\mathrm{deck}}$.  For the second, put
\begin{equation}\label{eq:GKT-residue-group}
 4\widetilde M\subset M\subset\widetilde M,
 \qquad
 Q=\widetilde M/4\widetilde M\simeq(\ZZ/4)^2;
\end{equation}
here $4\widetilde M$ has index two in $M$.  We call $Q$ the \emph{twist
group}: the class of $F(a,b)$ in $Q$ is the pair $(a,b)$ modulo $4$, so that
under the dictionary $Q$ is the group of GKT twists $(a,b)$ modulo $4$
(Section~\ref{subsec:gkt-recollections}).  The monodromy about each singular
point acts on $Q$ by the order-four matrix
\begin{equation}\label{eq:residue-matrix}
 P_{\mathrm{res}}=\begin{pmatrix}2&1\\3&0\end{pmatrix}\pmod4,
\end{equation}
the core holonomy acts by $K\equiv P_{\mathrm{res}}^2$, and the boundary
holonomies act trivially.  The \emph{twist cover} is the connected cover of
$\Acal^\circ$ on which the local system $Q$ becomes constant, that is, the
cover defined by the kernel of the monodromy action on $Q$.  It has degree
four, the order of $P_{\mathrm{res}}$, and since $4\widetilde M\subset M$ it
factors through $p_{\mathrm{fr}}$.  The transported Kummer roots of the
annular coefficient system (Definition~\ref{def:pathwise-root-object}), the
volume line and the wall structure pull back to both covers.  The hyperbolic
exponent holonomy persists on both: the first closed core lift to the twist
cover acts by $K^2$.

Since the ground field has characteristic zero, the idempotents
\[
 e_+=\frac{1+\delta_{\mathrm{deck}}}{2},
 \qquad e_-=\frac{1-\delta_{\mathrm{deck}}}{2}
\]
split the pushforward along $p_{\mathrm{fr}}$ of a deck-linearized object on
the fractional-exponent cover into its $\delta_{\mathrm{deck}}$-invariant and
$\delta_{\mathrm{deck}}$-anti-invariant summands;
Remark~\ref{rem:reflection-deck} applies this splitting to the transported
Jacobian algebras.
